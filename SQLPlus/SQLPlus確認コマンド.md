SHOW PDBS;

SHOW CON_NAME;

ALTER SESSION SET CONTAINER = ORCLPDB1;

コンテナ一覧
SELECT CON_ID, NAME, DBID FROM V$CONTAINERS;

PDBとサービスの対応
SELECT NAME, PDB FROM V$SERVICES;
